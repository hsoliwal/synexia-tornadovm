cmake_minimum_required(VERSION 3.20)
project(tornado_diffusion C)
find_package(JNI REQUIRED)
add_library(tornado_diffusion SHARED tornado_diffusion.c tornado_diffusion_jni.c)
target_include_directories(tornado_diffusion PRIVATE ${JNI_INCLUDE_DIRS})
if (UNIX AND NOT APPLE)
  target_link_libraries(tornado_diffusion PRIVATE m)
endif()
set_target_properties(tornado_diffusion PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES POSITION_INDEPENDENT_CODE YES)
